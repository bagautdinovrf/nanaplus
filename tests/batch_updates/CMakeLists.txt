cmake_minimum_required(VERSION 3.20)
project(nanaplus_batch_updates_tests LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# Build the actual library from this checkout, independently of any consumer.
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." "${CMAKE_CURRENT_BINARY_DIR}/nana")
target_compile_definitions(nana PUBLIC NANA_IGNORE_CONF)
if(MSVC)
    target_compile_options(nana PRIVATE /utf-8 /Zc:__cplusplus /permissive-)
endif()

add_executable(batch_updates_tests main.cpp)
target_link_libraries(batch_updates_tests PRIVATE nana::nana)
if(WIN32)
    target_link_libraries(batch_updates_tests PRIVATE user32 gdi32 comdlg32 shell32 ole32 uuid comctl32)
endif()
if(MSVC)
    target_compile_options(batch_updates_tests PRIVATE /utf-8 /Zc:__cplusplus /permissive-)
endif()

enable_testing()
foreach(scenario self_refresh mutual_refresh nested_batch cross_root exception_recovery)
    add_test(NAME batch_updates.${scenario} COMMAND batch_updates_tests ${scenario})
    set_tests_properties(batch_updates.${scenario} PROPERTIES TIMEOUT 30)
endforeach()
if(WIN32)
    add_test(NAME batch_updates.native_event COMMAND batch_updates_tests native_event)
    set_tests_properties(batch_updates.native_event PROPERTIES TIMEOUT 30)
    add_test(NAME batch_updates.native_event_inside_batch COMMAND batch_updates_tests native_event_inside_batch)
    set_tests_properties(batch_updates.native_event_inside_batch PROPERTIES TIMEOUT 30)
endif()
